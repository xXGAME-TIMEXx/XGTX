cmake_minimum_required(VERSION 3.16)
project(xgtx_exchange_tests LANGUAGES C CXX)
enable_testing()
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
set(EXCHANGE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../..")
set(LMDB_ROOT "${EXCHANGE_ROOT}/external/db_drivers/liblmdb")
add_library(xgtx_test_lmdb STATIC "${LMDB_ROOT}/mdb.c" "${LMDB_ROOT}/midl.c")
target_include_directories(xgtx_test_lmdb PUBLIC "${LMDB_ROOT}")
target_link_libraries(xgtx_test_lmdb PUBLIC Threads::Threads)
add_executable(xgtx_exchange_tests test_engine.cpp "${EXCHANGE_ROOT}/src/xgtx_exchange/engine.cpp")
target_compile_features(xgtx_exchange_tests PRIVATE cxx_std_17)
target_compile_options(xgtx_exchange_tests PRIVATE -Wall -Wextra -Werror)
target_link_libraries(xgtx_exchange_tests PRIVATE xgtx_test_lmdb OpenSSL::Crypto)
add_test(NAME xgtx_exchange COMMAND xgtx_exchange_tests)
